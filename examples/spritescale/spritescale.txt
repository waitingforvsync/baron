t   = 0 ..< ncam
sn  = SIN(t*k)
cs  = COS(t*k)
den = 1 + sn*sn
campath = camorig + ZIP(cs/den, FULL(ncam, 0), SIN(2*t*k)/(2*den)) * radius
dxdt = sn*(sn*sn - 3)/den^2
dzdt = (1 - 3*sn*sn)/den^2
hdg  = normalise(ZIP(dxdt, FULL(ncam, 0), dzdt))
target = hdg*lookahead + {0, -camorig[1], 0}
fwd    = normalise(target)
right  = ZIP(hdg[.., 2], FULL(ncam, 0), -hdg[.., 0])
up     = cross(fwd, right)
rel = objects[.., 0..2] - ZIP(campath)
vx  = viewdot(rel, right)
vy  = viewdot(rel, up)
vz  = viewdot(rel, fwd)
vzc = MAX({vz, near}, 0)

sxc   = focal*vx/vzc / 2 + 64
syp   = (127 - (64 - focal*vy/vzc)) / 2

xlb   = nearest(sxc - wcols/2)
syb   = nearest(syp)
scale8 = sizeb*256 DIV sprw
pwx    = (sprw*scale8 + 128) DIV 256
phpx    = (sprh*scale8 + 128) DIV 256
